local function split(str, sep)
    local t = {}
    for s in string.gmatch(str, "([^" .. sep .. "]+)") do
        table.insert(t, s)
    end
    return t
end
local safe = function(f, ...) return pcall(f, ...) end

-- ===== Anti Kick =====
local oldKick
oldKick = hookmetamethod(LocalPlayer, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    if tostring(method):lower() == "kick" then
        return nil
    end
    return oldKick(self, ...)
end)
hookfunction(LocalPlayer.Kick, function(...) return nil end)
local mt = getrawmetatable(game)
setreadonly(mt, false)
local oldIndex = mt.__index
mt.__index = newcclosure(function(self, key)
    if tostring(key):lower() == "kick" then
        return function() return nil end
    end
    return oldIndex(self, key)
end)

local function findHitboxEvent()
    local char = LocalPlayer.Character
    if not char then return nil end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return nil end
    for _, v in ipairs(tool:GetDescendants()) do
        if v:IsA("RemoteEvent") and v.Name == "Hitbox" then
            return v
        end
    end
    return nil
end

    Default=false,
    Callback=function(state)

        task.spawn(function()
            while fastAttack do
                task.wait()
                local remote = findHitboxEvent()
                if remote then pcall(function() remote:FireServer(8,1) end) end
            end
        end)
    end
})

    table.insert(dropdown.Options, plr.Name)
    dropdown:Refresh(dropdown.Options)

    for i, name in ipairs(dropdown.Options) do
        if name == plr.Name then
            table.remove(dropdown.Options, i)
            dropdown:Refresh(dropdown.Options)
            if selectedPlayerName == plr.Name then selectedPlayerName = nil end
            break
        end
    end
end)

-- Lấy tên tool / sword / fruit / melee player đang dùng
local function getPlayerItems(plr)
    local items = {}
    local char = plr.Character
    if not char then return "" end

    -- Tool / Sword đang cầm
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then table.insert(items, tool.Name) end

    -- Armor mặc trên người
    for _, armor in ipairs({
        "Cursed-Armor","Unique-Armor","Darkness-Armor","Thunder-Armor",
        "Diamond-Armor","Golden-Armor","Epic-Armor","Iron-Armor","Wooden-Armor"
    }) do
        if char:FindFirstChild(armor) then
            table.insert(items, armor)
        end
    end

    -- Lấy toàn bộ Tool trong Backpack (Devil Fruit / Sword / Item)
    if plr:FindFirstChild("Backpack") then
        for _, v in ipairs(plr.Backpack:GetChildren()) do
            if v:IsA("Tool") and not table.find(items, v.Name) then
                table.insert(items, v.Name)
            end
        end
    end

    -- Lấy thêm Tool trong Character (Melee, Sword, Fruit, Item khác)
    for _, v in ipairs(char:GetChildren()) do
        if v:IsA("Tool") and not table.find(items, v.Name) then
            table.insert(items, v.Name)
        end
    end

    return table.concat(items, ", ")
end

-- Tạo hoặc cập nhật ESP
local function updateESP(plr)
    if plr == LocalPlayer then return end
    local char = plr.Character
    if not char then return end
    local head = char:FindFirstChild("Head")
    if not head then return end

    local itemsText = getPlayerItems(plr)

    if espLabels[plr] then
        local label = espLabels[plr]:FindFirstChildOfClass("TextLabel")
        if label then
            label.Text = plr.Name .. "\n" .. itemsText
        end
    else
        local billboard = Instance.new("BillboardGui")
        billboard.Size = UDim2.new(0,200,0,100)
        billboard.Adornee = head
        billboard.AlwaysOnTop = true
        billboard.Name = "ESPLabel"

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.TextColor3 = Color3.new(1,0,0)
        label.TextStrokeTransparency = 0
        label.Font = Enum.Font.SourceSansBold
        label.TextScaled = true
        label.Text = plr.Name .. "\n" .. itemsText
        label.Parent = billboard

        billboard.Parent = PlayerGui
        espLabels[plr] = billboard
    end
end

local function removeESP(plr)
    if espLabels[plr] then
        espLabels[plr]:Destroy()
        espLabels[plr] = nil
    end
end

-- Hàm tìm skill remote
local function findSkillRemote(player, item, key)
    if not player then return nil end
    local containers = {player.Character, player:FindFirstChildOfClass("Backpack")}
    for _, container in ipairs(containers) do
        if container then
            for _, tool in ipairs(container:GetChildren()) do
                if tool:IsA("Tool") and string.find(tool.Name, item) then
                    local skillFolder = tool:FindFirstChild(key)
                    if skillFolder then
                        for _, ev in ipairs(skillFolder:GetChildren()) do
                            if ev:IsA("RemoteEvent") then
                                return ev
                            end
                        end
                    end
                end
            end
        end
    end
    return nil
end

            end
        end
    })
